cmake_minimum_required(VERSION 3.16)
project(databaze_kontaktu LANGUAGES CXX)
find_package(LibXml2 REQUIRED)
foreach(varianta sablony dedicnost)
    add_executable(kontakty_${varianta} ${varianta}/main.cpp)
    target_compile_features(kontakty_${varianta} PRIVATE cxx_std_17)
    target_include_directories(kontakty_${varianta} PRIVATE spolecne)
    target_link_libraries(kontakty_${varianta} PRIVATE LibXml2::LibXml2)
    target_compile_definitions(kontakty_${varianta} PRIVATE KONTAKTY_DIR="${CMAKE_CURRENT_SOURCE_DIR}/kontakty")
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(kontakty_${varianta} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    enable_testing()
    add_test(NAME import_kontaktu COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/overeni.py
        $<TARGET_FILE:kontakty_sablony> $<TARGET_FILE:kontakty_dedicnost>)
endif()
